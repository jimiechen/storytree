继续沿用上一轮修复后的架构，不再发明新机制。

```text
ShapeToolbar
  ├─ 圆柱体按钮（已有）
  ├─ 长方体按钮（已有）
  └─ 光源按钮（新增）           ← 调 store.addLight()
  └─ 主题切换按钮（新增）       ← 调 store.toggleTheme()

shot-3d-store
  ├─ cylinders（已有）
  ├─ boxes（已有）
  ├─ lights（新增）             ← LightObject[]
  ├─ selectedObjectId（已有，统一池）
  └─ theme（新增）：'light' | 'dark'

ThreeViewport
  ├─ cylinder-manager（已有）
  ├─ box-manager（已有）
  ├─ light-manager（新增）      ← 与 cylinder-manager 同构
  ├─ theme-controller（新增）   ← 监听 theme 变化更新场景

  ├─ Effect 2：TransformControls 绑定（在三类 entries 中查找）
  └─ Effect 3：theme 同步

```text

（直接对 LineSegments 做 TransformControls 也可以，

    boxes: s.boxes.map(b => ({ ...b, selected: b.id === id })),
    lights: s.lights.map(l => ({ ...l, selected: l.id === id })),
  }));
};

  setScene('theme', t => (t === 'dark' ? 'light' : 'dark'));
};

};

## **三、新增 light-manager.ts**

跟 `cylinder-manager.ts` 同构，关键区别在 mesh 的构造。

```ts
// novel-3d/components/ThreeViewport/light-manager.ts
import * as THREE from 'three';
import { createSignal, type Accessor } from 'solid-js';
import type { LightObject } from '../../shot-3d-store';

export interface LightMeshEntry {
  id: string;
  mesh: THREE.Object3D;   // 用 Group 作为 transform target
}

export interface LightManager {
  group: THREE.Group;
  entries: Accessor<LightMeshEntry[]>;
  update: (lights: LightObject[], selectedId?: string, theme?: 'light' | 'dark') => void;
  dispose: () => void;
  raycast: (raycaster: THREE.Raycaster) => LightMeshEntry | null;
  findById: (id: string) => LightMeshEntry | undefined;
}

interface LightVisual {
  root: THREE.Group;        // TransformControls 的目标
  cone: THREE.LineSegments; // 虚线圆锥
  hit: THREE.Mesh;          // 透明 mesh，用于 raycast
  material: THREE.LineDashedMaterial;
}

function createLightVisual(l: LightObject): LightVisual {
  const root = new THREE.Group();
  root.userData.id = l.id;
  root.userData.kind = 'light';

  const radius = Math.tan((l.angleDeg * Math.PI) / 180) * l.range;
  const coneGeom = new THREE.ConeGeometry(radius, l.range, 24, 1, true);

  // ConeGeometry 的尖端朝 +Y，这里把它翻成"尖端在原点、底面朝 -Y"
  // 这样把 root 放在光源位置，cone 表示从光源向下的光照锥体
  coneGeom.translate(0, -l.range / 2, 0);

  const edges = new THREE.EdgesGeometry(coneGeom);
  const material = new THREE.LineDashedMaterial({
    color: new THREE.Color(l.color),
    dashSize: 0.08,
    gapSize: 0.05,
    transparent: true,
    opacity: Math.min(1, 0.5 + l.intensity * 0.3),
  });
  const cone = new THREE.LineSegments(edges, material);
  cone.computeLineDistances();   // 必须！否则 dashed 不生效

  // 透明 hit mesh：与圆锥同形，仅供 raycaster 命中
  const hitMat = new THREE.MeshBasicMaterial({
    visible: false,            // 不渲染但仍参与 raycast
    side: THREE.DoubleSide,
  });
  const hit = new THREE.Mesh(coneGeom.clone(), hitMat);
  hit.userData.id = l.id;
  hit.userData.kind = 'light';

  // 顶部加一个小球作为"光源点"视觉提示
  const bulbGeom = new THREE.SphereGeometry(0.08, 12, 8);
  const bulbMat = new THREE.MeshBasicMaterial({ color: l.color });
  const bulb = new THREE.Mesh(bulbGeom, bulbMat);

  root.add(cone);
  root.add(hit);
  root.add(bulb);

  // 释放时归还原始几何
  root.userData.dispose = () => {
    coneGeom.dispose();
    edges.dispose();
    material.dispose();
    bulbGeom.dispose();
    bulbMat.dispose();
    (hit.geometry as THREE.BufferGeometry).dispose();
    hitMat.dispose();
  };

  return { root, cone, hit, material };
}

export function createLightManager(): LightManager {
  const group = new THREE.Group();
  group.name = 'LightGroup';

  const [entries, setEntries] = createSignal<LightMeshEntry[]>([]);
  const visualMap = new Map<string, LightVisual>();

  const update = (lights: LightObject[], selectedId?: string, theme: 'light' | 'dark' = 'dark') => {
    const incoming = new Set(lights.map(l => l.id));

    // 删除
    for (const [id, v] of visualMap) {
      if (!incoming.has(id)) {
        group.remove(v.root);
        (v.root.userData.dispose as () => void)?.();
        visualMap.delete(id);
      }
    }

    // 新增 / 更新
    for (const l of lights) {
      let v = visualMap.get(l.id);
      const isNew = !v;
      if (!v) {
        v = createLightVisual(l);
        group.add(v.root);
        visualMap.set(l.id, v);
      }

      // 颜色 / 透明度
      v.material.color.set(l.color);
      v.material.opacity = Math.min(1, 0.5 + l.intensity * 0.3);

      // 选中高亮
      const isSelected = l.id === selectedId;
      v.material.color.set(isSelected ? '#ffaa00' : l.color);
      v.material.dashSize = isSelected ? 0.12 : 0.08;
      v.material.needsUpdate = true;

      // 主题适配：浅色画布把虚线 opacity 调高一点，避免看不清
      if (theme === 'light') {
        v.material.opacity = Math.min(1, v.material.opacity + 0.2);
      }

      // 位置 / 旋转：epsilon 保护
      const targetX = l.position.x;
      const targetY = l.position.y;
      const targetZ = l.position.z;
      const dp =
        Math.abs(v.root.position.x - targetX) +
        Math.abs(v.root.position.y - targetY) +
        Math.abs(v.root.position.z - targetZ);
      if (isNew || dp > 1e-4) {
        v.root.position.set(targetX, targetY, targetZ);
      }
      const dr =
        Math.abs(v.root.rotation.x - l.rotation.x) +
        Math.abs(v.root.rotation.y - l.rotation.y) +
        Math.abs(v.root.rotation.z - l.rotation.z);
      if (isNew || dr > 1e-4) {
        v.root.rotation.set(l.rotation.x, l.rotation.y, l.rotation.z);
      }
    }

    // 一次性提交 entries
    const next: LightMeshEntry[] = [];
    for (const l of lights) {
      const v = visualMap.get(l.id);
      if (v) next.push({ id: l.id, mesh: v.root });
    }
    setEntries(next);
  };

  const raycast = (raycaster: THREE.Raycaster): LightMeshEntry | null => {
    // 只对 hit mesh 做 intersect，避免 LineSegments 命中难
    const candidates: THREE.Object3D[] = [];
    for (const v of visualMap.values()) candidates.push(v.hit);
    const hits = raycaster.intersectObjects(candidates, false);
    if (hits.length === 0) return null;
    const id = hits[0].object.userData?.id;
    if (!id) return null;
    const v = visualMap.get(id);
    return v ? { id, mesh: v.root } : null;
  };

  const findById = (id: string) => entries().find(e => e.id === id);

  const dispose = () => {
    for (const [, v] of visualMap) {
      (v.root.userData.dispose as () => void)?.();
    }
    visualMap.clear();
    setEntries([]);
    group.clear();
  };

  return { group, entries, update, dispose, raycast, findById };
}
```

要点：

```text
1. 用 Group 作为 root，TransformControls 直接绑 root，cone/bulb/hit 跟着走
2. 虚线必须调 computeLineDistances()
3. 透明 hit mesh 让 raycast 命中稳定，userData.id 保留以便反查
4. 与 cylinder-manager 完全同构：entries 是 Signal、用 Map diff、epsilon 保护
```

### **1. 不要去动 CSS body 背景**

```ts
// novel-3d/components/ThreeViewport/theme-controller.ts
import * as THREE from 'three';
import type { Theme } from '../../shot-3d-store';

interface ThemePalette {
  background: number;
  gridMain: number;
  gridSub: number;
  ambient: number;
  ambientIntensity: number;
  hemiSky: number;
  hemiGround: number;
}

const PALETTES: Record<Theme, ThemePalette> = {
  dark: {
    background: 0x1a1a1f,
    gridMain: 0x666666,
    gridSub: 0x333333,
    ambient: 0xffffff,
    ambientIntensity: 0.35,
    hemiSky: 0x88aaff,
    hemiGround: 0x202020,
  },
  light: {
    background: 0xf5f5f0,
    gridMain: 0x888888,
    gridSub: 0xcccccc,
    ambient: 0xffffff,
    ambientIntensity: 0.7,
    hemiSky: 0xffffff,
    hemiGround: 0xb0b0b0,
  },
};

export interface ThemeController {
  apply: (theme: Theme) => void;
}

export function createThemeController(
  scene: THREE.Scene,
  refs: {
    grid: THREE.GridHelper;
    ambient: THREE.AmbientLight;
    hemi: THREE.HemisphereLight;
    renderer: THREE.WebGLRenderer;
  }
): ThemeController {
  const apply = (theme: Theme) => {
    const p = PALETTES[theme];

    scene.background = new THREE.Color(p.background);
    refs.renderer.setClearColor(p.background, 1);

    // GridHelper 有两组 material（中心轴 + 网格）
    const mats = Array.isArray(refs.grid.material) ? refs.grid.material : [refs.grid.material];
    if (mats[0] instanceof THREE.LineBasicMaterial) (mats[0] as THREE.LineBasicMaterial).color.setHex(p.gridMain);
    if (mats[1] instanceof THREE.LineBasicMaterial) (mats[1] as THREE.LineBasicMaterial).color.setHex(p.gridSub);

    refs.ambient.color.setHex(p.ambient);
    refs.ambient.intensity = p.ambientIntensity;
    refs.hemi.color.setHex(p.hemiSky);
    refs.hemi.groundColor.setHex(p.hemiGround);
  };

  return { apply };
}
```

## 四、light-manager.ts 要求

1. 与 cylinder-manager 同构：
   - entries 是 Solid Signal
   - 用 Map<id, LightVisual> 做 diff，禁止 dispose-recreate
   - 每个 root.userData = { id, kind: 'light', dispose: () => void }
   - update() 末尾一次性 setEntries

2. LightVisual 由三部分组成：
   - root: THREE.Group（TransformControls 的目标）
   - cone: LineSegments(EdgesGeometry(ConeGeometry), LineDashedMaterial)
   - bulb: SphereGeometry 小球，使用光源颜色
   - hit: 透明 MeshBasicMaterial(visible=false) 的 Mesh，仅用于 raycast

3. ConeGeometry 必须 translate(0, -range/2, 0)，让锥尖在 root 原点，底面朝 -Y。

4. cone.computeLineDistances() 必须调用，否则 LineDashedMaterial 不显示虚线。

5. update 中 position/rotation 必须使用 epsilon 保护：
   ```
   if (isNew || dp > 1e-4) v.root.position.set(...);

## **八、几个值得提前考虑的细节**

第一个是 **光源的“真实光照”要不要现在做**。建议现在不要做。当前阶段只把光源当作“可视化标记 + 镜头规划数据”，真正的渲染光（SpotLight / DirectionalLight）等到 Storyboard / 渲染插件再加。理由是一旦加了真实光，性能、阴影、PBR 材质都会跟着卷进来，MVP 阶段没必要。

第二个是 **角度参数 angleDeg 后续怎么编辑**。InfoPanel 应该在选中类型为 light 时显示 color、intensity、range、angleDeg 四个滑块。这次先不实现编辑面板，但 store 已经留好字段，下一轮 PR 可以直接接上。

第三个是 **主题切换不要等动画**。建议主题切换瞬时生效，因为 Three.js 渲染本身在持续刷新，颜色突变并不突兀。如果你想做柔和过渡，可以在 theme-controller 里用 `THREE.Color.lerp` 做 200ms 的插值，但这是 nice-to-have，不要堵在主流程上。

第四个是 **导出/序列化**。光源加进 store 之后，所有“保存场景到 JSON / 加载场景 JSON”的逻辑要把 lights 字段一并带上，主题字段也要序列化（用户上次用浅色，下次打开仍然是浅色）。这个可以放在你下一个 PR 里收尾。

第五个是 **未来 3D Shot Draft 插件的复用**。light-manager 和 theme-controller 这次写完之后，把它们抽成 `packages/creative-runtime-3d/` 下的通用模块会更好——和 cylinder-manager / box-manager 一起，构成 3D Shot Draft 插件的渲染层基础。这样后续接相机、角色、道具时不用重新造一遍 Manager 范式。
