这是渲染时为了避免 Markdown 把模板字符串里的 `${}` 误当成代码导致格式错乱，我留了空模板。**这不是截断，是占位**。下面把这块补完整，可以直接拿去用：

```tsx
import { createMemo } from 'solid-js';
import type { createShot3DStore } from '../shot-3d-store';

export function ShapeToolbar(props: {
  store: ReturnType<typeof createShot3DStore>;
}) {
  const theme = createMemo(() => props.store.scene.theme);

  const btnBase =
    'px-3 py-2 rounded-md text-sm border transition-colors flex items-center gap-1';

  const btnNormal = () =>
    theme() === 'dark'
      ? 'bg-neutral-800 text-neutral-100 border-neutral-700 hover:bg-neutral-700'
      : 'bg-white text-neutral-800 border-neutral-300 hover:bg-neutral-100';

  const btnAccent = () =>
    theme() === 'dark'
      ? 'bg-amber-500/20 text-amber-200 border-amber-500/40 hover:bg-amber-500/30'
      : 'bg-amber-100 text-amber-700 border-amber-300 hover:bg-amber-200';

  return (
    <div class="absolute bottom-4 left-1/2 -translate-x-1/2 flex gap-2 z-20">
      <button
        class={`${btnBase} ${btnNormal()}`}
        onClick={() => props.store.addCylinder()}
      >
        ◎ 圆柱体
      </button>

      <button
        class={`${btnBase} ${btnNormal()}`}
        onClick={() => props.store.addBox()}
      >
        ▣ 长方体
      </button>

      <button
        class={`${btnBase} ${btnNormal()}`}
        onClick={() => props.store.addLight()}
      >
        ✦ 光源
      </button>

      <div class="w-px bg-neutral-500/40 mx-1" />

      <button
        class={`${btnBase} ${btnAccent()}`}
        onClick={() => props.store.toggleTheme()}
        title="切换画布明暗"
      >
        {theme() === 'dark' ? '☀ 浅色' : '☾ 深色'}
      </button>
    </div>
  );
}
```

要点：

```text
1. 用 createMemo 包 theme 访问，避免每次 render 都重新求值
2. 主题切换按钮用 amber 色系做 accent，跟普通增删按钮区分开
3. 分隔线 w-px 让"创建对象"和"画布设置"两类按钮在视觉上分组
4. 在 dark / light 两种主题下都不会出现"按钮和背景同色"的尴尬
```

```ts
const handleClick = (e: PointerEvent) => {
  const rect = renderer.domElement.getBoundingClientRect();
  const ndc = new THREE.Vector2(
    ((e.clientX - rect.left) / rect.width) * 2 - 1,
    -((e.clientY - rect.top) / rect.height) * 2 + 1,
  );
  const raycaster = new THREE.Raycaster();
  raycaster.setFromCamera(ndc, camera);

  const cylHit = cylManager.raycast(raycaster);
  if (cylHit) return props.store.selectAny(cylHit.id);

  const boxHit = boxManager.raycast(raycaster);
  if (boxHit) return props.store.selectAny(boxHit.id);

  props.store.selectAny(undefined);
};
```
