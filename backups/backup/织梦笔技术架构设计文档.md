### 1.2 设计原则

| 原则 | 说明 | 落地策略 |
|------|------|---------|
| **客户端优先** | 核心编辑体验在本地完成，减少网络依赖 | VS Code OSS Fork 提供本地编辑器，AI 功能异步请求 |
| **离线可用** | 基础写作功能离线可用，AI 功能在线使用 | 本地文件存储 + 在线 AI 服务 |
| **渐进增强** | 从 Extension 开始，逐步深入 Fork 定制 | v1.0 以内置 Extension 为主，v2.0 深入 Fork |
| **模型无关** | 不绑定特定 AI 模型，支持热插拔 | 统一模型适配层，BYOK 模式 |
| **数据主权** | 用户作品数据用户所有，平台不窥探 | 端到端加密、本地优先存储 |
| **可观测性** | 全链路追踪、日志、指标 | OpenTelemetry + Prometheus + Grafana |
| **全球优先** | 多区域部署、CDN 加速、延迟路由，服务全球用户 | AWS 多区域部署 + CloudFront CDN + Cloudflare 延迟路由 |
| **合规优先** | 满足 GDPR/CCPA/COPPA 等国际隐私法规 | 数据本地化、Cookie Consent、隐私 API、年龄验证 |

### 2.1 VS Code OSS Fork 策略

#### 2.1.1 Fork 深度与路径

采用**渐进式 Fork 策略**，参考 Cursor 的做法：

```
阶段一（v1.0）：浅度 Fork + 内置 Extension
  ├── 修改 product.json（品牌化、遥测端点）
  ├── 替换图标和资源文件
  ├── 移除不需要的内置功能（调试器、Git 等）
  ├── 在 extensions/ 中添加内置 Extension
  └── 修改默认 Workbench 布局

阶段二（v1.5）：中度 Fork + 核心 UI 定制
  ├── 修改 Activity Bar 布局（写作场景定制）
  ├── 自定义侧边栏默认视图
  ├── 修改 Monaco Editor 默认配置
  ├── 添加写作场景特有的编辑器贡献
  └── 扩展 Extension API（添加写作专用 API）

阶段三（v2.0）：深度 Fork + 编辑器深度定制
  ├── 自定义 TextModel（支持章节/分支元数据）
  ├── 自定义编辑器装饰（行内 AI 补全、一致性标记）
  ├── 自定义 Diff Provider（分支对比）
  └── 性能深度优化（大文件、虚拟滚动）
```

#### 2.3.4 新增的 Extension API

在 `src/vscode-dts/vscode.proposed.dreamweaver.d.ts` 中定义写作专用 API：

```typescript
declare module 'vscode' {

  // ========== AI 写作 API ==========

  /** AI 写作服务 */
  export namespace dreamweaver {

    // --- 模型管理 ---
    interface ModelInfo {
      id: string;
      name: string;
      provider: string;
      capabilities: ModelCapability[];
      contextWindow: number;
      pricing: { input: number; output: number };
    }

    interface ModelCapability {
      type: 'continuation' | 'expansion' | 'rewriting' | 'dialogue'
           | 'description' | 'reasoning' | 'polishing';
      score: number; // 0-10
    }

    /** 获取可用模型列表 */
    function getAvailableModels(): Thenable<ModelInfo[]>;

    /** 获取当前默认模型 */
    function getCurrentModel(): Thenable<ModelInfo>;

    /** 设置当前模型 */
    function setCurrentModel(modelId: string): Thenable<void>;

    // --- AI 生成 ---
    interface AIGenerationOptions {
      modelId?: string;
      temperature?: number;
      maxTokens?: number;
      systemPrompt?: string;
      context?: AIContextOptions;
      stream?: boolean;
    }

    interface AIContextOptions {
      includeCharacters?: string[];   // 包含哪些角色信息
      includeLocations?: string[];    // 包含哪些地点信息
      includeForeshadowing?: boolean;  // 包含伏笔信息
      includeRecentChapters?: number;  // 包含最近N章
      customContext?: string;          // 自定义上下文
    }

    interface AIGenerationResult {
      text: string;
      modelId: string;
      tokensUsed: { input: number; output: number };
      consistencyWarnings: ConsistencyWarning[];
    }

    /** AI 续写 */
    function continueWriting(
      options?: AIGenerationOptions
    ): Thenable<AIGenerationResult>;

    /** AI 扩写选中文本 */
    function expandText(
      selection: string,
      options?: AIGenerationOptions
    ): Thenable<AIGenerationResult>;

    /** AI 改写选中文本 */
    function rewriteText(
      selection: string,
      instruction: string,
      options?: AIGenerationOptions
    ): Thenable<AIGenerationResult>;

    /** AI 生成对话 */
    function generateDialogue(
      characters: string[],
      context: string,
      options?: AIGenerationOptions
    ): Thenable<AIGenerationResult>;

    /** AI 生成场景描写 */
    function generateDescription(
      scene: string,
      style?: string,
      options?: AIGenerationOptions
    ): Thenable<AIGenerationResult>;

    /** 流式 AI 生成 */
    function streamGenerate(
      prompt: string,
      options?: AIGenerationOptions,
      callback: (chunk: string) => void
    ): Thenable<AIGenerationResult>;

    // --- 多模型协作 ---
    interface CollaborationOptions {
      models: string[];
      mode: 'vote' | 'relay' | 'debate' | 'merge' | 'review';
      prompt: string;
      options?: AIGenerationOptions;
    }

    interface CollaborationResult {
      results: { modelId: string; text: string }[];
      finalText?: string;
      votes?: { modelId: string; chosen: boolean }[];
    }

    function collaborativeGenerate(
      options: CollaborationOptions
    ): Thenable<CollaborationResult>;

    // --- 一致性检查 ---
    interface ConsistencyWarning {
      type: 'character' | 'timeline' | 'setting' | 'logic' | 'style';
      severity: 'info' | 'warning' | 'error';
      message: string;
      location: { line: number; column: number };
      suggestion: string;
    }

    function checkConsistency(
      text: string,
      workId: string
    ): Thenable<ConsistencyWarning[]>;

    // --- 去AI味 ---
    interface DeAIResult {
      text: string;
      originalScore: number;  // 原始AI味评分 0-100
      processedScore: number; // 处理后AI味评分 0-100
      changes: { original: string; replacement: string }[];
    }

    function removeAITone(
      text: string,
      intensity?: 'light' | 'medium' | 'heavy'
    ): Thenable<DeAIResult>;

    // --- 知识库 ---
    interface StoryBibleEntry {
      id: string;
      type: 'character' | 'location' | 'item' | 'faction'
           | 'timeline' | 'foreshadowing' | 'worldRule' | 'styleGuide';
      name: string;
      content: Record<string, any>;
      tags: string[];
      relatedEntries: string[];
    }

    function getStoryBibleEntries(
      workId: string,
      type?: string
    ): Thenable<StoryBibleEntry[]>;

    function updateStoryBibleEntry(
      workId: string,
      entry: StoryBibleEntry
    ): Thenable<void>;

    function autoExtractEntities(
      workId: string,
      chapterId: string
    ): Thenable<StoryBibleEntry[]>;

    // --- 分支管理 ---
    interface BranchInfo {
      id: string;
      name: string;
      parentId: string | null;
      type: 'main' | 'sub' | 'if' | 'archived';
      depth: number;
      chapterCount: number;
      wordCount: number;
      isActive: boolean;
    }

    function getBranches(workId: string): Thenable<BranchInfo[]>;

    function createBranch(
      workId: string,
      parentNodeId: string,
      name: string,
      type?: BranchInfo['type']
    ): Thenable<BranchInfo>;

    function switchBranch(workId: string, branchId: string): Thenable<void>;

    function mergeBranch(
      workId: string,
      sourceBranchId: string,
      targetBranchId: string,
      options?: { sections?: string[] }
    ): Thenable<void>;

    // --- 作品管理 ---
    interface WorkInfo {
      id: string;
      title: string;
      genre: string;
      totalWords: number;
      chapterCount: number;
      status: 'draft' | 'serializing' | 'completed';
      currentBranchId: string;
    }

    function getCurrentWork(): Thenable<WorkInfo | null>;

    function openWork(workId: string): Thenable<void>;

    // --- 事件 ---
    interface DreamWeaverEvents {
      onModelChanged: Event<ModelInfo>;
      onAIStreamChunk: Event<string>;
      onConsistencyWarning: Event<ConsistencyWarning>;
      onBranchChanged: Event<BranchInfo>;
      onWordCountChanged: Event<{ total: number; session: number }>;
    }

    export const events: DreamWeaverEvents;
  }
}
```

```typescript
import LanguageDetector from 'i18next-browser-languagedetector';
import HttpBackend from 'i18next-http-backend';

i18n
  .use(HttpBackend)               // 从 CDN 加载翻译文件
  .use(LanguageDetector)           // 自动检测用户语言
  .use(initReactI18next)           // 绑定 React
  .init({
    fallbackLng: 'en',             // 默认语言
    supportedLngs: ['en', 'zh', 'ja', 'ko', 'es', 'fr', 'de'],
    debug: process.env.NODE_ENV === 'development',

    detection: {
      // 语言检测优先级
      order: [
        'querystring',             // ?lng=en
        'localStorage',            // 用户手动切换的语言
        'navigator',               // 浏览器语言设置
        'htmlTag',                 // <html lang="en">
      ],
      caches: ['localStorage'],    // 缓存用户选择
      lookupQuerystring: 'lng',
      lookupLocalStorage: 'dreamweaver-lang',
    },

    backend: {
      // 翻译文件从 CDN 加载，支持按需加载
      loadPath: '/locales/{{lng}}/{{ns}}.json',
    },

    interpolation: {
      escapeValue: false,          // React 已自动转义
    },

    react: {
      useSuspense: true,           // 使用 Suspense 模式
    },
  });
```

### 3.2 服务间通信

| 通信方式 | 使用场景 | 技术实现 |
|---------|---------|---------|
| **同步 HTTP/gRPC** | 服务间直接调用 | Go 服务间用 gRPC，跨语言用 REST |
| **异步消息队列** | 耗时任务（AI生成、一致性检查、向量化） | RabbitMQ |
| **事件驱动** | 状态变更通知（章节保存→触发一致性检查） | RabbitMQ Event |
| **共享数据库** | 少量需要事务的场景 | PostgreSQL（避免分布式事务） |

**消息队列 Topic 设计**：

```
dw.chapter.saved        → 触发一致性检查 + 实体提取 + 向量化
dw.chapter.updated      → 触发增量向量化
dw.branch.created       → 通知分支服务更新索引
dw.branch.merged        → 触发合并后一致性检查
dw.ai.generation.request → AI 引擎消费生成请求
dw.ai.generation.result  → 生成结果通知
dw.consistency.check.request → 一致性检查请求
dw.consistency.check.result  → 检查结果通知
```

### 4.2 模型适配器设计

```python
# 模型适配器基类
from abc import ABC, abstractmethod
from typing import AsyncIterator, Optional

class BaseModelAdapter(ABC):
    """所有模型适配器的基类"""

    @abstractmethod
    async def generate(
        self,
        messages: list[dict],
        options: GenerateOptions
    ) -> GenerateResult:
        """同步生成"""
        pass

    @abstractmethod
    async def stream(
        self,
        messages: list[dict],
        options: GenerateOptions
    ) -> AsyncIterator[str]:
        """流式生成"""
        pass

    @abstractmethod
    async def count_tokens(self, text: str) -> int:
        """Token 计数"""
        pass

    @abstractmethod
    def get_capabilities(self) -> ModelCapabilities:
        """获取模型能力"""
        pass

    @abstractmethod
    async def health_check(self) -> bool:
        """健康检查"""
        pass


# Claude 适配器示例
class ClaudeAdapter(BaseModelAdapter):
    """Anthropic Claude 模型适配器"""

    def __init__(self, api_key: str, model: str = "claude-4-opus"):
        self.client = AsyncAnthropic(api_key=api_key)
        self.model = model

    async def generate(self, messages, options):
        response = await self.client.messages.create(
            model=self.model,
            max_tokens=options.max_tokens or 4096,
            temperature=options.temperature or 0.7,
            system=self._extract_system(messages),
            messages=self._extract_messages(messages),
        )
        return GenerateResult(
            text=response.content[0].text,
            model_id=self.model,
            tokens_used={
                "input": response.usage.input_tokens,
                "output": response.usage.output_tokens,
            }
        )

    async def stream(self, messages, options):
        async with self.client.messages.stream(
            model=self.model,
            max_tokens=options.max_tokens or 4096,
            temperature=options.temperature or 0.7,
            system=self._extract_system(messages),
            messages=self._extract_messages(messages),
        ) as stream:
            async for text in stream.text_stream:
                yield text

    async def count_tokens(self, text: str) -> int:
        # Claude 使用近似计数
        return len(text) // 2  # 中文约2字符/token

    def get_capabilities(self) -> ModelCapabilities:
        return ModelCapabilities(
            model_id=self.model,
            provider="anthropic",
            context_window=200000,
            strengths=["creative_writing", "long_context", "dialogue"],
            creative_writing_score=9.5,
            chinese_quality_score=8.5,
        )


# 模型适配器工厂
class ModelAdapterFactory:
    """模型适配器工厂"""

    _adapters: dict[str, type[BaseModelAdapter]] = {
        "claude-4-opus": ClaudeAdapter,
        "claude-4-sonnet": ClaudeAdapter,
        "gpt-4o": GPTAdapter,
        "gpt-4o-mini": GPTAdapter,
        "deepseek-v3": DeepSeekAdapter,
        "qwen-max": QwenAdapter,
        "ernie-4.0": ErnieAdapter,
        "gemini-2.5-pro": GeminiAdapter,
        "moonshot-v1": KimiAdapter,
    }

    @classmethod
    def create(cls, model_id: str, api_key: str) -> BaseModelAdapter:
        adapter_cls = cls._adapters.get(model_id)
        if not adapter_cls:
            raise ValueError(f"Unsupported model: {model_id}")
        return adapter_cls(api_key=api_key, model=model_id)

    @classmethod
    def register(cls, model_id: str, adapter_cls: type[BaseModelAdapter]):
        """注册自定义模型适配器（BYOK）"""
        cls._adapters[model_id] = adapter_cls
```

| **MinIO** | 对象存储（导出文件、封面图片、备份） | 大文件、二进制 |

```
  └── Branch: if_B (IF线B，从 Ch3 分叉)
        ├── Chapter 3 (IF线B版本)
        └── Chapter 4B (IF线B独有)
```

#### 6.1.2 分支创建算法

```python
async def create_branch(
    work_id: str,
    parent_chapter_id: str,
    branch_name: str,
    branch_type: str = "sub"
) -> Branch:
    """
    在指定章节处创建新分支

    算法：
    1. 找到 parent_chapter 所属的当前分支
    2. 复制 parent_chapter 及之前所有章节的引用到新分支
    3. parent_chapter 本身创建一个副本（可独立编辑）
    4. 新分支从 parent_chapter 副本开始独立发展
    """

    # 获取父章节信息
    parent_chapter = await chapter_repo.get(parent_chapter_id)
    parent_branch_id = parent_chapter.branch_id

    # 创建新分支
    new_branch = Branch(
        work_id=work_id,
        parent_branch_id=parent_branch_id,
        parent_node_id=parent_chapter_id,
        name=branch_name,
        type=branch_type,
        depth=parent_branch.depth + 1,
    )
    await branch_repo.create(new_branch)

    # 复制父分支中 parent_chapter 及之前的章节引用
    previous_chapters = await chapter_repo.get_by_branch_before(
        branch_id=parent_branch_id,
        chapter_number=parent_chapter.chapter_number
    )

    for ch in previous_chapters:
        if ch.id == parent_chapter_id:
            # 分叉点章节：创建副本
            fork_chapter = Chapter(
                work_id=work_id,
                branch_id=new_branch.id,
                title=ch.title,
                chapter_number=ch.chapter_number,
                content=ch.content,  # 复制内容
                outline=ch.outline,
                notes=ch.notes,
                status=ch.status,
            )
            await chapter_repo.create(fork_chapter)
        else:
            # 分叉点之前的章节：共享引用（不复制）
            # 通过 branch_chapter_mapping 表建立映射
            await branch_chapter_mapping_repo.create(
                branch_id=new_branch.id,
                chapter_id=ch.id,
                chapter_number=ch.chapter_number,
                is_shared=True,
            )

    return new_branch
```

#### 6.1.3 分支合并算法

```python
async def merge_branch(
    source_branch_id: str,
    target_branch_id: str,
    merge_options: MergeOptions
) -> MergeResult:
    """
    合并源分支到目标分支

    策略：
    1. 找到两个分支的共同祖先（分叉点）
    2. 对比分叉点之后的差异
    3. 执行三方合并
    4. 标记冲突（如有）
    """

    # 找到共同祖先
    source_branch = await branch_repo.get(source_branch_id)
    common_ancestor_id = source_branch.parent_node_id

    # 获取分叉点之后两个分支的章节
    source_chapters = await chapter_repo.get_after_fork(
        branch_id=source_branch_id,
        fork_point_id=common_ancestor_id
    )
    target_chapters = await chapter_repo.get_after_fork(
        branch_id=target_branch_id,
        fork_point_id=common_ancestor_id
    )

    # AI 辅助合并分析
    merge_analysis = await ai_engine.analyze_merge(
        source_content=serialize_chapters(source_chapters),
        target_content=serialize_chapters(target_chapters),
        common_ancestor_content=await get_chapter_content(common_ancestor_id),
    )

    # 执行合并
    conflicts = []
    for source_ch in source_chapters:
        # 检查目标分支是否有对应章节
        target_ch = find_matching_chapter(target_chapters, source_ch)

        if target_ch is None:
            # 目标分支没有对应章节 → 直接追加
            await append_chapter_to_branch(target_branch_id, source_ch)
        else:
            # 两分支都有该章节 → 需要合并
            merge_result = await three_way_merge(
                base=common_ancestor_content,
                source=source_ch.content,
                target=target_ch.content,
            )
            if merge_result.has_conflicts:
                conflicts.append(MergeConflict(
                    chapter_number=source_ch.chapter_number,
                    source_content=source_ch.content,
                    target_content=target_ch.content,
                    conflict_regions=merge_result.conflict_regions,
                    ai_suggestion=merge_analysis.suggestions.get(
                        source_ch.chapter_number
                    ),
                ))
            else:
                # 无冲突，自动合并
                await update_chapter(target_ch.id, content=merge_result.merged)

    return MergeResult(
        merged_count=len(source_chapters) - len(conflicts),
        conflict_count=len(conflicts),
        conflicts=conflicts,
    )
```

### 6.2 一致性检查引擎

```python
class ConsistencyEngine:
    """一致性检查引擎"""

    def __init__(self, bible_service, ai_service):
        self.bible_service = bible_service
        self.ai_service = ai_service

    async def check_chapter(
        self, work_id: str, chapter_id: str, content: str
    ) -> list[ConsistencyWarning]:
        """对章节内容进行全面一致性检查"""

        warnings = []

        # 并行执行各类检查
        results = await asyncio.gather(
            self.check_character_consistency(work_id, content),
            self.check_timeline_consistency(work_id, content),
            self.check_setting_consistency(work_id, content),
            self.check_foreshadowing(work_id, chapter_id, content),
            self.check_style_consistency(work_id, content),
        )

        for result in results:
            warnings.extend(result)

        # 按严重程度排序
        severity_order = {"error": 0, "warning": 1, "info": 2}
        warnings.sort(key=lambda w: severity_order[w.severity])

        return warnings

    async def check_character_consistency(
        self, work_id: str, content: str
    ) -> list[ConsistencyWarning]:
        """角色一致性检查"""

        # 1. 从内容中提取提及的角色
        mentioned_characters = await self._extract_characters(content)

        # 2. 获取知识库中的角色设定
        characters = await self.bible_service.get_entries(
            work_id, entry_type="character"
        )
        character_map = {c.name: c for c in characters}

        warnings = []
        for char_name in mentioned_characters:
            if char_name not in character_map:
                continue

            char_setting = character_map[char_name]

            # 3. AI 检查内容中角色行为是否与设定一致
            check_result = await self.ai_service.generate(
                system_prompt=CHARACTER_CHECK_PROMPT,
                messages=[
                    {"role": "user", "content": f"""
                    角色设定: {json.dumps(char_setting.content, ensure_ascii=False)}
                    当前内容: {content}

                    请检查当前内容中角色"{char_name}"的行为、语言、外貌描述
                    是否与设定一致。如不一致，请指出矛盾点。
                    """}
                ],
            )

            if check_result.consistency_issues:
                for issue in check_result.consistency_issues:
                    warnings.append(ConsistencyWarning(
                        type="character",
                        severity=issue.severity,
                        message=f"角色「{char_name}」: {issue.description}",
                        location=issue.location,
                        suggestion=issue.suggestion,
                    ))

        return warnings

    async def check_foreshadowing(
        self, work_id: str, chapter_id: str, content: str
    ) -> list[ConsistencyWarning]:
        """伏笔追踪检查"""

        warnings = []

        # 获取所有未回收的伏笔
        foreshadowings = await self.bible_service.get_entries(
            work_id, entry_type="foreshadowing"
        )
        unresolved = [f for f in foreshadowings if f.content.get("status") == "pending"]

        # 检查当前章节是否适合回收某些伏笔
        for fs in unresolved:
            # 检查距离埋设的章节数
            chapters_since_plant = await self._count_chapters_between(
                work_id, fs.source_chapter_id, chapter_id
            )

            if chapters_since_plant > 20:
                warnings.append(ConsistencyWarning(
                    type="character",  # 归类为角色相关
                    severity="warning",
                    message=f"伏笔「{fs.name}」已埋设 {chapters_since_plant} 章未回收",
                    suggestion="考虑在近期章节中回收此伏笔，避免读者遗忘",
                ))

            # AI 检查当前内容是否暗示了伏笔回收
            relevance = await self.ai_service.generate(
                system_prompt=FORESHADOW_CHECK_PROMPT,
                messages=[
                    {"role": "user", "content": f"""
                    伏笔内容: {fs.name} - {fs.content.get('description')}
                    当前章节内容: {content}

                    当前章节是否暗示或回收了此伏笔？
                    """}
                ],
            )

            if relevance.is_resolved:
                # 标记伏笔为已回收
                await self.bible_service.update_entry(
                    work_id, fs.id,
                    content={**fs.content, "status": "resolved",
                             "resolved_chapter_id": chapter_id}
                )

        return warnings
```

### 12.2 Stripe Checkout 集成

```python
# 后端：创建 Checkout Session
from stripe import Stripe

async def create_checkout_session(
    user_id: str,
    price_id: str,  # Stripe Price ID (e.g., price_pro_monthly)
    success_url: str,
    cancel_url: str,
) -> str:
    """创建 Stripe Checkout Session"""

    session = await stripe.checkout.sessions.create_async(
        mode="subscription",
        payment_method_types=["card"],
        line_items=[{"price": price_id, "quantity": 1}],
        success_url=success_url,
        cancel_url=cancel_url,
        metadata={
            "user_id": user_id,
            "app": "dreamweaver",
        },
        subscription_data={
            "trial_period_days": 7,  # 7 天免费试用
            "metadata": {"user_id": user_id},
        },
    )

    return session.url
```

### 12.3 Webhook 事件处理

```python
# Stripe Webhook 处理器
from stripe import Webhook

STRIPE_WEBHOOK_SECRET = config.stripe_webhook_secret

@app.post("/api/v1/webhooks/stripe")
async def stripe_webhook(request: Request):
    payload = await request.body()
    sig_header = request.headers["stripe-signature"]

    # 验证 Webhook 签名
    event = Webhook.construct_event(
        payload, sig_header, STRIPE_WEBHOOK_SECRET
    )

    event_handlers = {
        "payment_intent.succeeded": handle_payment_succeeded,
        "invoice.paid": handle_invoice_paid,
        "invoice.payment_failed": handle_invoice_payment_failed,
        "customer.subscription.created": handle_subscription_created,
        "customer.subscription.updated": handle_subscription_updated,
        "customer.subscription.deleted": handle_subscription_deleted,
        "customer.subscription.trial_will_end": handle_trial_ending,
    }

    handler = event_handlers.get(event["type"])
    if handler:
        await handler(event["data"]["object"])

    return JSONResponse(status_code=200, content={"received": True})


async def handle_subscription_deleted(subscription):
```

```python
# 用量计量服务
class UsageMeteringService:
    """用量计量：字数统计 + AI Token 统计"""

    async def record_word_count(
        self, user_id: str, work_id: str, words: int
    ):
        """记录用户写作字数"""
        await usage_repo.create(
            user_id=user_id,
            resource_type="word_count",
            amount=words,
            unit="words",
            metadata={"work_id": work_id},
        )

    async def record_ai_tokens(
        self, user_id: str, tokens: int, model_id: str
    ):
        """记录 AI Token 消耗"""
        await usage_repo.create(
            user_id=user_id,
            resource_type="ai_tokens",
            amount=tokens,
            unit="tokens",
            metadata={"model_id": model_id},
        )

    async def get_usage_summary(
        self, user_id: str, period: str = "current_month"
    ) -> UsageSummary:
        """获取用量汇总（用于 Stripe Billing 报告）"""
        return await usage_repo.get_summary(user_id, period)

    async def report_to_stripe(self, user_id: str):
        """向 Stripe 报告用量（用于按量计费）"""
        summary = await self.get_usage_summary(user_id)
        await stripe.subscription_items.create_usage_record_async(
            subscription_item_id=summary.stripe_subscription_item_id,
            quantity=summary.total_ai_tokens // 1000,  # 按 1000 tokens 计量
            timestamp=int(time.time()),
            action="increment",
        )
```

### 13.3 GDPR 数据删除 API（级联删除）

```python
# GDPR 数据删除 API - 级联删除所有关联数据
@app.delete("/api/v1/privacy/data-erasure")
async def erase_user_data(
    user_id: str = Depends(get_current_user),
    confirmation: str = Body(..., embed=True),
):
    """
    GDPR Right to Erasure - 被遗忘权
    级联删除用户所有数据及其关联数据
    """

    # 二次确认
    if confirmation != "PERMANENTLY_DELETE_MY_DATA":
        raise HTTPException(400, "Invalid confirmation token")

    deletion_log = []

    # 1. 删除用户作品及关联数据
    works = await work_service.get_all_works(user_id)
    for work in works:
        # 删除章节内容
        chapters = await chapter_service.get_by_work(work.id)
        for chapter in chapters:
            await chapter_version_service.delete_by_chapter(chapter.id)
            await chapter_service.delete(chapter.id)
            deletion_log.append(f"Chapter {chapter.id} deleted")

        # 删除知识库条目及向量
        entries = await bible_service.get_all_entries(work.id)
        for entry in entries:
            await milvus_service.delete(entry.embedding_id)  # 删除向量
            await bible_version_service.delete_by_entry(entry.id)
            await bible_service.delete(entry.id)
            deletion_log.append(f"Bible entry {entry.id} deleted")

        # 删除分支
        await branch_service.delete_by_work(work.id)
        deletion_log.append(f"Branches for work {work.id} deleted")

        # 删除一致性检查记录
        await consistency_service.delete_by_work(work.id)

        # 删除作品
        await work_service.delete(work.id)
        deletion_log.append(f"Work {work.id} deleted")

    # 2. 删除 AI 生成记录
    await ai_service.delete_all_generations(user_id)
    deletion_log.append("AI generation history deleted")

    # 3. 删除用量记录
    await usage_service.delete_all_records(user_id)
    deletion_log.append("Usage records deleted")

    # 4. 删除支付历史
    await payment_service.anonymize(user_id)  # 支付记录匿名化（法规要求保留）
    deletion_log.append("Payment history anonymized")

    # 5. 删除 Stripe 客户（取消订阅）
    await stripe_service.delete_customer(user_id)
    deletion_log.append("Stripe customer deleted")

    # 6. 删除用户账户
    await user_service.delete(user_id)
    deletion_log.append("User account deleted")

    # 7. 记录删除审计日志（保留 30 天）
    await audit_log_service.record_deletion(
        user_id=user_id,
        details=deletion_log,
        retention_days=30,
    )

    return {"status": "completed", "deleted_items": len(deletion_log)}
```

### 13.4 Cookie Consent 管理

```typescript
// 前端 Cookie Consent 组件
// src/shared/components/CookieConsent.tsx

interface CookiePreferences {
  necessary: boolean;    // 必要 Cookie（始终开启）
  analytics: boolean;    // 分析 Cookie（Sentry、Datadog）
  marketing: boolean;    // 营销 Cookie
  preferences: boolean;  // 偏好 Cookie（语言、主题）
}

const COOKIE_CONSENT_KEY = 'dreamweaver-cookie-consent';

export function CookieConsent() {
  const { t } = useTranslation('compliance');
  const [showBanner, setShowBanner] = useState(false);

  useEffect(() => {
    const consent = localStorage.getItem(COOKIE_CONSENT_KEY);
    if (!consent) {
      setShowBanner(true);
    } else {
      applyConsent(JSON.parse(consent));
    }
  }, []);

  function applyConsent(preferences: CookiePreferences) {
    // 根据用户选择启用/禁用对应的 Cookie 和追踪
    if (preferences.analytics) {
      Sentry.init({ enabled: true });
      datadogRum.init({ enabled: true });
    } else {
      Sentry.init({ enabled: false });
      datadogRum.init({ enabled: false });
    }
  }

  function acceptAll() {
    const preferences: CookiePreferences = {
      necessary: true,
      analytics: true,
      marketing: true,
      preferences: true,
    };
    localStorage.setItem(COOKIE_CONSENT_KEY, JSON.stringify(preferences));
    applyConsent(preferences);
    setShowBanner(false);
  }

  function rejectNonEssential() {
    const preferences: CookiePreferences = {
      necessary: true,
      analytics: false,
      marketing: false,
      preferences: false,
    };
    localStorage.setItem(COOKIE_CONSENT_KEY, JSON.stringify(preferences));
    applyConsent(preferences);
    setShowBanner(false);
  }

  if (!showBanner) return null;

  return (
    <div className="cookie-consent-banner">
      <h3>{t('cookieConsent.title')}</h3>
      <p>{t('cookieConsent.description')}</p>
      <button onClick={acceptAll}>{t('cookieConsent.accept')}</button>
      <button onClick={rejectNonEssential}>{t('cookieConsent.reject')}</button>
      <button onClick={() => setShowSettings(true)}>{t('cookieConsent.settings')}</button>
    </div>
  );
}
```
